# Gateway dependencies. Single source of truth for both the Dockerfile and the host
# .venv-gateway used for local debugging.
#
# litellm floor is a SECURITY control, not a preference: 1.82.7 and 1.82.8 were malicious
# releases (TeamPCP PyPI compromise, 2026-03-24) shipping a .pth payload that executed on
# every Python process start. Safe ranges are <=1.82.6 or >=1.83.0.
litellm[proxy]>=1.85.1,<2

# fastapi ceiling is a COMPATIBILITY pin, bisected by hand against this litellm version.
# litellm 1.95.0 declares `fastapi<1.0,>=0.136.3`, but that upper bound is wrong: fastapi
# removed `fastapi.dependencies.utils.get_flat_dependant`, which litellm's management
# endpoints import at startup, so the proxy dies with an ImportError before it ever binds
# a port. Bisected boundary: present through 0.140.6, gone from 0.140.7 onward.
# Revisit when litellm fixes its own range. The backend is unaffected — it tracks current
# fastapi independently, which is another reason these two live in separate images.
fastapi<0.140.7
